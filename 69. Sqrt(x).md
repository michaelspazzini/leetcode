class Solution:
    def mySqrt(self, x: int) -> int:
        if x < 2:
            return x
        
        left = 1
        right = x//2
        while left <= right:
            mid = (left+right)//2
            mid2 = mid*mid

            if mid2 == x:
                return mid
            elif mid2 < x:
                left = mid + 1 
            else:
                right = mid - 1

        return right
```

class Solution:
    def mySqrt(self, x: int) -> int:
        if x == 0:
            return 0
        ok = []
        for i in range(1,x+1):
            tmp = i*i
            if tmp < x:
                ok.append(i)
            elif tmp == x:
                return i
            else:
                return max(ok)
```
