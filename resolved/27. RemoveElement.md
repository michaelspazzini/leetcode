```python3
class Solution:
    def removeElement(self, nums: List[int], val: int) -> int:

        lp = 0
        rp = len(nums)-1

        while lp <= rp:
            if nums[lp] == val:
                nums[lp] = nums[rp]
                nums[rp] = val
                rp -= 1
            else:
                lp += 1

        return lp
```

```c
int removeElement(int *nums, int numsSize, int val) {
    int *ptr = nums;
    int k = 0;
    while (numsSize > 0) {
        numsSize--;
        *ptr = *nums;
        if (*nums != val) {
            ptr++;
            k++;
        }
        nums++;
    }
    return k;
}
```
