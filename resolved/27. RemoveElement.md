```python3
class Solution:
    def removeElement(self, nums: List[int], val: int) -> int:

        lp = 0
        rp = len(nums)-1

        while lp <= rp:
            if nums[lp] == val:
                nums[lp] = nums[rp]
                nums[rp] = val
                rp -= 1
            else:
                lp += 1

        return lp
```
