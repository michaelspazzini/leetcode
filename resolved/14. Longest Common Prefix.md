```python3
class Solution:
    def longestCommonPrefix(self, strs: List[str]) -> str:

        def checkChars(chars):
            first = chars[0]
            for c in chars:
                if c != first:
                    return False
            return True

        words=[x for x in strs]
        maxLengthReadable = len(words[0])
        i=0
        result=""
        for word in words:
            if len(word) < maxLengthReadable:
                maxLengthReadable = len(word)

        while i < maxLengthReadable:
            char=[x[i] for x in strs]
            print(char)
            if checkChars(char):
                result+=char[0]
            else:
                break
            i=i+1
        
        return result
```
