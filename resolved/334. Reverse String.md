```c
void reverseString(char *s, int sSize) {
    size_t i = 0;
    size_t j = 0;
    size_t ssize = sSize;
    char tmp = 0;
    char other = 0;
    __asm__ volatile (
        "mov $0, %[ai];"
        "mov %[asize], %[aj];"
        "dec %[aj];"
    "1:"
        "cmp %[aj], %[ai];"
        "jae 2f;"
        "mov (%[as], %[ai]), %[atmp];"
        "mov (%[as], %[aj]), %[aother];"
        "mov %[aother], (%[as], %[ai]);"
        "mov %[atmp], (%[as], %[aj]);"
        "inc %[ai];"
        "dec %[aj];"
        "jmp 1b;"
    "2:"
        :
        [ai] "+r" (i),
        [aj] "+r" (j),
        [atmp] "+r" (tmp),
        [aother] "+r" (other)
        :
        [asize] "r" (ssize),
        [as] "r" (s)
        :
        "cc", "memory"
    );
}
```
