```python3 []
class Solution:
    def generate(self, numRows: int) -> List[List[int]]:
        out=[]
        pos=0
        for i in range (0,numRows):
            out += [[0] * (i+1)]
            out[i][0] = out[i][-1] = 1

            for j in range (1, len(out[i])-1):
               out[i][j] = out[i-1][j-1] + out[i-1][j]

        return out
```
