```python3 []
class Solution:
    def generate(self, numRows: int) -> List[List[int]]:
        out=[]
        pos=0
        for i in range (0,numRows):
            out += [[0] * (i+1)]
            out[i][0] = out[i][-1] = 1

            for j in range (1, len(out[i])-1):
               out[i][j] = out[i-1][j-1] + out[i-1][j]

        return out
```

```python3
class Solution(object):
    def generate(self, numRows):
        """
        :type numRows: int
        :rtype: List[List[int]]
        """

        if numRows == 0:
            return []
        if numRows == 1:
            return [[1]]
        if numRows == 2:
            return [[1], [1,1]]

        pascal = [[1], [1,1]]
        temp = []

        for i in range(2, numRows):
            temp.append(pascal[i-1][0])
            for j in range(1, i):
                temp.append(pascal[i-1][j-1]+pascal[i-1][j])
            temp.append(pascal[i-1][len(pascal[i-1])-1])

            pascal.append(temp)
            temp = []

        return pascal
```
