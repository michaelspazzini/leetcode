```python []
class Solution:
    def getCommon(self, nums1: List[int], nums2: List[int]) -> int:
        lp=0
        rp=0
        maxLp = len(nums1)-1
        maxRp = len(nums2)-1
        while nums1[lp] != nums2[rp]:
            if nums1[lp] < nums2[rp]:
                lp +=1
            elif nums1[lp] > nums2[rp]:
                rp +=1
            if lp > maxLp or rp > maxRp:
                return -1
            
        
        if nums1[lp] == nums2[rp]:
            return nums1[lp]
        else:
            return -1
```
