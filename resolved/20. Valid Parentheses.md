## Soluzione 1
- Runtime: 3 ms
- Memory: 19.36 MB

```python3
class Solution:
    def isValid(self, s: str) -> bool:
        stack = []
        
        mapping = {
            ')': '(',
            ']': '[',
            '}': '{'
        }
        
        for char in s:
            if char in mapping.values():
                stack.append(char)
            else:
                if not stack:
                    return False
                if stack[-1] != mapping[char]:
                    return False
                stack.pop()
        
        return len(stack) == 0
            
```

## Soluzione 2 (x86_64 assembly)

This is a direct translation of the python solution, but implemented in x86_64 assembly.  The code is wrapped in C functions because leetcode doesn't support assembly programs directly.

```c
// Author: Andrea Fiori

__attribute__((naked))
long mapping(long c) {
    __asm__(".intel_syntax noprefix;"
"        mov rax, 40;"
"        cmp rdi, 41;"
"        je _ret%=;"
"        mov rax, 91;"
"        cmp rdi, 93;"
"        je _ret%=;"
"        mov rax, 123;"
"        cmp rdi, 125;"
"        je _ret%=;"
"        mov rax, 0;"
"        _ret%=:"
"        ret;"
".att_syntax prefix;" : :);
}

__attribute__((naked))
long isOpening(long c) {
__asm__(".intel_syntax noprefix;"
// 40 = (
"        cmp rdi, 40;"
"        je _equal%=;"
// 91 = [
"        cmp rdi, 91;"
"        je _equal%=;"
// 123 = {
"        cmp rdi, 123;"
"        je _equal%=;"

"        mov rax, 0;"
"        ret;"

"        _equal%=:"
"        mov rax, 1;"
"        ret;"
".att_syntax prefix;" : :);
}

__attribute__((naked))
bool isValid(char* s) {
    __asm__(".intel_syntax noprefix;"
"        push rbp;"
"        mov rbp, rsp;"
"        sub rsp, 48;"

// save str len and s to local variables
"        mov [rbp - 8], rdi;"
"        call strlen;"
"        mov [rbp - 16], rax;"

// allocate stack
"        mov rdi, rax;"
"        call malloc;"
"        mov qword ptr [rbp - 24], rax;"

// stack ptr local var
"        mov qword ptr [rbp - 32], 0;"

// string ptr local var
"        mov qword ptr [rbp - 40], 0;"

// main for loop
"        _for%=:"

"        mov r8, qword ptr [rbp - 40];"
"        cmp r8, qword ptr [rbp - 16];"
"        je _endfor%=;"

// r8 = char
"        mov r9, qword ptr [rbp - 8];"
"        mov r10, qword ptr [rbp - 40];"
"        movzx r8, byte ptr [r9 + r10];"

"        inc qword ptr [rbp - 40];"

"        mov rdi, r8;"
"        call isOpening;"

"        cmp rax, 0;"
"        je _else%=;"
// append to stack
"        mov r9, qword ptr [rbp - 32];"
"        mov r10, qword ptr [rbp - 24];"
"        mov byte ptr [r10 + r9], r8b;"
"        inc qword ptr [rbp - 32];"
"        jmp _endif%=;"
"        _else%=:"
"        cmp qword ptr [rbp - 32], 0;"
"        je _retfalse%=;"
"        mov r9, qword ptr [rbp - 32];"
"        mov r10, qword ptr [rbp - 24];"
"        movzx r9, byte ptr [r9 + r10 - 1];"
"        mov rdi, r8;"
"        call mapping;"
"        cmp r9, rax;"
"        jne _retfalse%=;"
// stack.pop()
"        dec qword ptr [rbp - 32];"
"        _endif%=:"
"        jmp _for%=;"
"        _endfor%=:"

"        cmp qword ptr [rbp - 32], 0;"
"        jne _retfalse%=;"

// return True
"        mov rax, 1;"
"        jmp _ret%=;"

"        _retfalse%=:"
"        mov rax, 0;"
"        jmp _ret%=;"

"        _ret%=:"
// free stack
"        push rax;"
"        mov rdi, qword ptr [rbp - 24];"
"        call free;"
"        pop rax;"
"        leave;"
"        ret;"
".att_syntax prefix;" : :);
}
```
