```python []
class Solution(object):
    def twoSum(self, nums, target):
        """
        :type nums: List[int]
        :type target: int
        :rtype: List[int]
        """
        hash = dict()
        for i,x in enumerate(nums):
            wanted = target - x
            if wanted in hash:
                return [hash[wanted], i]
            else:
                hash[x] = i

        return None
        
```

```c
    __asm__(
        ".intel_syntax noprefix;"

        "push rbp;"
        "mov rbp, rsp;"
        "sub rsp, 0x20;"

        "mov qword ptr [rbp - 8], rdi;" // nums
        "mov dword ptr [rbp - 12], esi;" // numsSize
        "mov dword ptr [rbp - 16], edx;" // target

        "mov dword ptr [rcx], 2;" // *returnSize = 2

        "mov edi, 8;"
        "call malloc;" // rax = result

        "xor r10d, r10d;" // i = 0

    "outer%=:"
        "cmp r10d, dword ptr [rbp - 12];"
        "jge endouter%=;"

        "lea r11d, [r10d + 1];" // j = i + 1

    "inner%=:"
        "cmp r11d, dword ptr [rbp - 12];"
        "jge endinner%=;"

        "mov r8, qword ptr [rbp - 8];" // r8 = nums

        "mov edx, dword ptr [r8 + r10*4];" // edx = nums[i]
        "add edx, dword ptr [r8 + r11*4];" // edx += nums[j]

        "cmp edx, dword ptr [rbp - 16];"
        "jne endif%=;"

        "mov dword ptr [rax], r10d;" // result[0] = i
        "mov dword ptr [rax + 4], r11d;" // result[1] = j
        "jmp endouter%=;"

    "endif%=:"
        "inc r11d;"
        "jmp inner%=;"

    "endinner%=:"
        "inc r10d;"
        "jmp outer%=;"

    "endouter%=:"
        "leave;"
        "ret;"

    ".att_syntax prefix;"
        : :);
}
```
